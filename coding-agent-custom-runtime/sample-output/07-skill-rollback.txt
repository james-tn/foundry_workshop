def detectAnomaly(this, window=24):
    """
    Robust z-score of the latest vibration reading against the trailing window.
    Team standard: anomaly-method-standard v1.
    """
    import numpy as np

    x = np.asarray(this.vibration, dtype=float)
    if x.size == 0:
        return {"method": "robust_z", "score": None, "isAnomaly": False}

    last = x[-1]
    ref = x[-window - 1:-1]
    if ref.size == 0:
        return {"method": "robust_z", "score": 0.0, "isAnomaly": False}

    med = np.median(ref)
    mad = np.median(np.abs(ref - med))
    if mad == 0:
        z = 0.0
    else:
        z = 0.6745 * (last - med) / mad

    return {"method": "robust_z", "score": round(float(z), 2), "isAnomaly": bool(abs(z) > 3.5)}
