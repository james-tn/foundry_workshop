def pctOverThreshold(this):
    readings = this.vibration or []
    if not readings:
        return None
    return sum(v > this.vibrationThreshold for v in readings) / len(readings)


def detectAnomalies(this, windowSize, zThreshold):
    import numpy as np
    readings = this.vibration or []
    if not readings:
        return []
    readings = np.array(readings, dtype=float)
    n = readings.size
    windowSize = int(windowSize)
    if windowSize <= 0:
        raise ValueError("windowSize must be positive")
    zThreshold = float(zThreshold)
    if n < windowSize:
        return []
    kernel = np.ones(windowSize) / windowSize
    means = np.convolve(readings, kernel, mode='valid')
    sq_means = np.convolve(readings * readings, kernel, mode='valid')
    stds = np.sqrt(np.maximum(sq_means - means * means, 0.0))
    pad = windowSize - 1
    anomalies = []
    for i in range(pad, n):
        j = i - pad
        std = stds[j]
        if std > 0.0 and abs((readings[i] - means[j]) / std) > zThreshold:
            anomalies.append(int(i))
    return anomalies
