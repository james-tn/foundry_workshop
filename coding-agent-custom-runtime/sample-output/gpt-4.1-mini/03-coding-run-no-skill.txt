def detect_vibration_anomalies(this):
    import numpy as np
    import ruptures as rpt

    readings = np.array(this.vibration or [])
    if len(readings) < 2:
        return []

    # Use Pelt algorithm with l2 cost to detect change points
    model = "l2"
    algo = rpt.Pelt(model=model).fit(readings)
    # max 5 change points allowed, can adjust penalty to tune sensitivity
    result = algo.predict(pen=3)

    # result contains change points indices (1-based), remove last element as it is length of signal
    change_points = result[:-1] if result else []
    return change_points
