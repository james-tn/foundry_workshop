def detectAnomaly(this, penalty=10.0, minShift=1.0):
    import numpy as np
    import ruptures as rpt

    x = np.asarray(this.vibration, dtype=float)
    bkps = rpt.Pelt(model="l2", min_size=3, jump=1).fit(x).predict(pen=penalty)
    cps = [b for b in bkps if b < len(x)]

    if not cps:
        return {"method": "pelt_l2", "changePointIndex": None, "levelShift": 0.0, "isAnomaly": False}

    i = cps[0]
    shift = float(x[i:].mean() - x[:i].mean())
    return {
        "method": "pelt_l2",
        "changePointIndex": int(i),
        "levelShift": round(shift, 2),
        "isAnomaly": bool(shift > minShift)
    }
