```python
def detectAnomaly(this, window=24):
    import numpy as np
    x = np.asarray(this.vibration, dtype=float)
    if len(x) == 0:
        return {"method": "robust_z", "score": 0.0, "isAnomaly": False}
    last = x[-1]
    ref = x[-window - 1:-1]
    if len(ref) == 0:
        return {"method": "robust_z", "score": 0.0, "isAnomaly": False}
    med = np.median(ref)
    mad = np.median(np.abs(ref - med))
    if mad == 0:
        z = 0.0
    else:
        z = 0.6745 * (last - med) / mad
    return {"method": "robust_z", "score": round(float(z), 2), "isAnomaly": bool(abs(z) > 3.5)}
```
